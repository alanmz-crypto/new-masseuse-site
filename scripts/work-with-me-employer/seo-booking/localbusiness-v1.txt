add_action( 'wp_head', function () {
    $home = untrailingslashit( set_url_scheme( home_url( '/' ), 'https' ) );
    if ( 'localhost' === wp_parse_url( $home, PHP_URL_HOST ) ) {
        $home = untrailingslashit( home_url( '/' ) ); // isolated HTTP rehearsal
    }
    $data = array(
        '@context' => 'https://schema.org',
        '@type' => 'LocalBusiness',
        '@id' => $home . '/#localbusiness',
        'name' => 'Thai Solitude',
        'telephone' => '+1-312-539-4328',
        'email' => 'butsabad@gmail.com',
        'image' => $home . '/wp-content/uploads/2026/09/cropped-cropped-butsaba_avatar-1.webp',
        'logo' => $home . '/wp-content/uploads/2026/09/cropped-cropped-butsaba_avatar-1.webp',
        'description' => 'Thai, therapeutic, deep tissue and chair massage in Portland with Butsaba Dusrai, LMT, plus corporate and group events. Every session starts with your goals.',
        'areaServed' => array( '@type' => 'City', 'name' => 'Portland' ),
        'founder' => array( '@id' => $home . '/#person' ),
        'address' => array(
            '@type' => 'PostalAddress',
            'addressLocality' => 'Portland',
            'addressRegion' => 'OR',
            'addressCountry' => 'US',
        ),
        'url' => $home,
    );
    echo '<script type="application/ld+json">' . wp_json_encode( $data, JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE ) . '</script>';
} );
